### 解题思路

- 本题经过分析：
  - 花费1：精灵球数量
  - 花费2：皮卡丘体力值
  - 价值：小精灵的数量

#### 双重价值背包

- 状态表示：
  - `f[i][j][k]`表示所有只从第`i`个物品中选，且花费1不超过`j`，花费2不超过`k`。
- 状态计算：
  - `f[i][j][k] = max(f[i][j][k], f[i - 1][j - v1[i]][k - v2[i]])`

### 实现代码


```cpp

#include <bits/stdc++.h>
using namespace std;

const int N = 1010, M = 510;
int n, V1, V2;
int f[N][M];

int main(){
    cin >> V1 >> V2 >> n;
    for(int i = 1; i <= n; i ++){
        int v1, v2;
        cin >> v1 >> v2;
        for(int j = V1; j >= v1; j --)
            for(int k = V2; k > v2; k --)
                f[j][k] = max(f[j][k], f[j - v1][k - v2] + 1);
    }
    
    int k = V2;
    
    while(k > 0 && f[V1][k] == f[V1][V2])
        k --;
    
    cout << f[V1][V2] << " " << V2 - k << endl;
    return 0;
}
```
