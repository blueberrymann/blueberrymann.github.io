### 解题思路

- 当 `qi == 0` 时表示这是一个跳板，需要增加人物能量：`e += v[idx]`
- 处理死循环问题：
  - 死循环产生原因：存在两个能量为0的跳板，人物在这两个跳板之间来回跳跃
  - 判断条件：如果再次经过某一跳板时，人物的能量没有变化，说明已进入死循环
  - 实现方式：使用 `energy[idx] == e` 判断，若相等则跳出循环

### 实现代码

```cpp
#include <bits/stdc++.h>
using namespace std;
const int N = 1e5 + 100;
int q[N], v[N];
int n, s;
bool state[N];
int energy[N];


int main(){
    cin >> n >> s;

    int min_val = 0x3f3f3f3f;
    bool flag = 1;

    for(int i = 1; i <= n; i ++){
        cin >> q[i] >> v[i];
        if(q[i] == 1)
            min_val = min(min_val, v[i]);
        else
            if(v[i] != 0)
                flag = 0;
    }
    int res = 0;
    if(flag){
        // 跳板全是0,加不上一点能量
        // 就要直接统计有多少炮弹能直接被击破
        for(int i = 1; i <= n ; i++){
            if(q[i] == 1 && s >= v[i])
                res ++;

        }

        cout << res << endl;
        return 0;
    }

    int idx = s;
    int e = 1;
    int direction = 1;

    while(idx <= n && idx >= 1){
        if(q[idx] == 0){
            // qi == 0为跳板
            e += v[idx];
            // 跳出死循环的关键条件（造成死循环的原因是有两个能量为0的跳板）
            // 如果再次经过某一跳板时，人的能量没有变化，那么即可判定已经进入死循环了
            if(energy[idx] == e)
            {
                break;
            }
            energy[idx] = e;
            direction *= -1;
            idx += direction * e;
        }else{
            if(e >= v[idx] && !state[idx]){
                res ++;
                state[idx] = true;
            }

            idx += direction * e;
        }

        // cout << "idx: " << idx << endl;
    }

    cout << res << endl;
    return 0;
}

```
